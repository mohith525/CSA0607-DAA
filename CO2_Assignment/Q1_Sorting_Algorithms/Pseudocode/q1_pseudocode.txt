Q1 - SORTING: PSEUDOCODE (comparisons = key comparisons, moves = swaps or shifts)

SelectionSort(A[0..n-1])
  for i <- 0 to n-2
      m <- i
      for j <- i+1 to n-1
          comparisons++
          if A[j] < A[m] then m <- j
      if m != i then swap(A[i], A[m]); moves++

BubbleSort(A[0..n-1])            // with early-exit flag
  for i <- 0 to n-2
      swapped <- false
      for j <- 0 to n-2-i
          comparisons++
          if A[j] > A[j+1] then swap(A[j], A[j+1]); moves++; swapped <- true
      if swapped = false then stop          // already sorted

InsertionSort(A[0..n-1])
  for i <- 1 to n-1
      v <- A[i];  j <- i-1
      while j >= 0
          comparisons++
          if A[j] > v then A[j+1] <- A[j]; moves++; j <- j-1
          else break
      A[j+1] <- v

TEST-DATA DESIGN (n = 200 and n = 1000, fixed random seed)
  A  Random permutation                                   (typical case)
  B  95% sorted: sorted block + 5% new random orders appended at the end (the fulfilment-centre scenario)
  C  Already sorted                                       (best case)
  D  Reverse sorted                                       (edge case: worst case)
  E  Only 5 distinct priority values (many duplicates)    (edge case)

PREDICTION (before running)
  Selection : comparisons always n(n-1)/2; at most n-1 swaps; NOT sensitive to input order.
  Bubble    : n(n-1)/2 comparisons on random/reverse; only n-1 on sorted (early exit).
  Insertion : comparisons about n(n-1)/4 on random, n(n-1)/2 on reverse, n-1 on sorted; work ~ number of inversions.
